#include <iostream>
#include <cmath>

using namespace std;

class NegativeNumber : public exception {
public:
    const char* what() const noexcept override {
        return "Negative Number Exception";
    }
};

int main() {
    int n;
    cout << "Enter Number: ";
    cin >> n;

    try {
        if (n < 0) {
            throw NegativeNumber();
        }
        double s = sqrt(n);
        cout << "Square root: " << s << endl;
    }
    catch ( NegativeNumber& e) {
        cout << e.what() << endl;
    }

    return 0;
}

{
    int x, y;
    char ch;
    cout << "Enter Numbers: ";
    cin >> x >> ch >> y;
    try
    {
        if (ch != '+' && ch != '-' && ch != '/' && ch != '*')
        {
            throw ch;
        }
        if (ch == '/' && y == 0)
        {
            throw 0;
        }
        if (ch == '+')
        {
            cout << x + y << endl;
        }
        else if (ch == '-')
        {
            cout << x - y << endl;
        }
        else if (ch == '/')
        {
            cout << x / y << endl;
        }
        else
        {
            cout << x * y << endl;
        }
    }
    catch (const char)
    {
        cout << "Invalid Operator" << '\n';
    }
    catch (const int)
    {
        cout << "Division by zero error" << endl;
    }
    return 0;

#include <iostream>
#include <fstream>
#include <string>
using namespace std;
int main()
{
    ifstream file("students.txt");
    if (!file)
    {
        cout << "file Could not be opened \n";
    }
    string line;
    string word;

    int characters{0};
    int words{0};
    int lines{0};
    // lines/char
    while (getline(file, line))
    {
        lines++;
        characters += line.length();
    }
    file.clear();
    file.seekg(0);
    while (file >> word)
    {
        words++;
    }
    file.close();
    cout << "Characters = " << characters << endl;
    cout << "Words = " << words << endl;
    cout << "Lines = " << lines << endl;

    return 0;
}
#include<iostream>
#include<fstream>
using namespace std;
int main(){
    ifstream source("source.txt");
    ofstream destination("destination.txt");
    if (!source)
    {
        cout<<"File Could not be opened \n";
    }
    char ch;
    while (source.get(ch))
    {
        destination.put(ch);
    }
    source.close();
    destination.close();
    cout<<"Copied Successfully \n";
    return 0;
    
}
